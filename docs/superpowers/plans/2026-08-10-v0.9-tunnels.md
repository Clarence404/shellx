# v0.9 SSH Tunnels Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add local-forward SSH tunnel support — users configure tunnel rules per host, choose a connection mode, and tunnels are established automatically on connect with live toggle support.

**Architecture:** The Rust backend gains a `tunnels` DB table, a `TunnelHandle` that binds a local `TcpListener` and splices each accepted TCP connection through an SSH `channel_open_direct_tcpip` channel, and six new IPC commands. The frontend gains a `TunnelsPanel` component shown as a session tab, and `HostForm` is split into Basic/Tunnels tabs with a connection-mode segmented control and an inline rule editor.

**Tech Stack:** Rust/Tauri 2, russh 0.62.5 (`channel_open_direct_tcpip`), rusqlite, tokio, React/TypeScript, Zustand, Vitest.

```rust
use crate::error::{Error, Result};
use rusqlite::{params, Connection};
use serde::{Deserialize, Serialize};
use std::sync::Arc;
use std::time::{SystemTime, UNIX_EPOCH};
use tokio::sync::Mutex;
use uuid::Uuid;

#[derive(Debug, Clone, Serialize)]
pub struct TunnelRule {
    pub id: Uuid,
    pub host_id: Uuid,
    pub label: String,
    pub local_port: u16,
    pub remote_host: String,
    pub remote_port: u16,
    pub enabled: bool,
    pub sort_order: i32,
    pub created_at: i64,
}

#[derive(Debug, Clone, Deserialize)]
pub struct NewTunnelRule {
    pub host_id: Uuid,
    pub label: String,
    pub local_port: u16,
    pub remote_host: String,
    pub remote_port: u16,
    pub enabled: Option<bool>,   // defaults to true
}

#[derive(Debug, Clone, Deserialize)]
pub struct UpdateTunnelRule {
    pub id: Uuid,
    pub label: Option<String>,
    pub local_port: Option<u16>,
    pub remote_host: Option<String>,
    pub remote_port: Option<u16>,
    pub enabled: Option<bool>,
    pub sort_order: Option<i32>,
}

pub struct TunnelStore {
    conn: Arc<Mutex<Connection>>,
}

impl TunnelStore {
    pub fn new(conn: Arc<Mutex<Connection>>) -> Self {
        Self { conn }
    }

    pub async fn list_for_host(&self, host_id: Uuid) -> Result<Vec<TunnelRule>> {
        let conn = self.conn.lock().await;
        let mut stmt = conn
            .prepare("SELECT id,host_id,label,local_port,remote_host,remote_port,enabled,sort_order,created_at FROM tunnels WHERE host_id=?1 ORDER BY sort_order,created_at")
            .map_err(|e| Error::Protocol(e.to_string()))?;
        let rows = stmt.query_map(params![host_id.to_string()], |row| {
            Ok(TunnelRule {
                id: row.get::<_,String>(0)?.parse().map_err(|_| rusqlite::Error::InvalidQuery)?,
                host_id: row.get::<_,String>(1)?.parse().map_err(|_| rusqlite::Error::InvalidQuery)?,
                label: row.get(2)?,
                local_port: row.get::<_,i64>(3)? as u16,
                remote_host: row.get(4)?,
                remote_port: row.get::<_,i64>(5)? as u16,
                enabled: row.get::<_,i64>(6)? != 0,
                sort_order: row.get::<_,i64>(7)? as i32,
                created_at: row.get(8)?,
            })
        }).map_err(|e| Error::Protocol(e.to_string()))?;
        rows.collect::<rusqlite::Result<Vec<_>>>().map_err(|e| Error::Protocol(e.to_string()))
    }

    pub async fn insert(&self, r: NewTunnelRule) -> Result<TunnelRule> {
        let id = Uuid::new_v4();
        let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs() as i64;
        let enabled = r.enabled.unwrap_or(true);
        let conn = self.conn.lock().await;
        conn.execute(
            "INSERT INTO tunnels (id,host_id,label,local_port,remote_host,remote_port,enabled,sort_order,created_at) VALUES (?1,?2,?3,?4,?5,?6,?7,0,?8)",
            params![id.to_string(), r.host_id.to_string(), r.label, r.local_port as i64, r.remote_host, r.remote_port as i64, enabled as i64, now],
        ).map_err(|e| Error::Protocol(e.to_string()))?;
        Ok(TunnelRule { id, host_id: r.host_id, label: r.label, local_port: r.local_port, remote_host: r.remote_host, remote_port: r.remote_port, enabled, sort_order: 0, created_at: now })
    }

    pub async fn update(&self, u: UpdateTunnelRule) -> Result<()> {
        let conn = self.conn.lock().await;
        if let Some(v) = u.label { conn.execute("UPDATE tunnels SET label=?1 WHERE id=?2", params![v, u.id.to_string()]).map_err(|e| Error::Protocol(e.to_string()))?; }
        if let Some(v) = u.local_port { conn.execute("UPDATE tunnels SET local_port=?1 WHERE id=?2", params![v as i64, u.id.to_string()]).map_err(|e| Error::Protocol(e.to_string()))?; }
        if let Some(v) = u.remote_host { conn.execute("UPDATE tunnels SET remote_host=?1 WHERE id=?2", params![v, u.id.to_string()]).map_err(|e| Error::Protocol(e.to_string()))?; }
        if let Some(v) = u.remote_port { conn.execute("UPDATE tunnels SET remote_port=?1 WHERE id=?2", params![v as i64, u.id.to_string()]).map_err(|e| Error::Protocol(e.to_string()))?; }
        if let Some(v) = u.enabled { conn.execute("UPDATE tunnels SET enabled=?1 WHERE id=?2", params![v as i64, u.id.to_string()]).map_err(|e| Error::Protocol(e.to_string()))?; }
        if let Some(v) = u.sort_order { conn.execute("UPDATE tunnels SET sort_order=?1 WHERE id=?2", params![v as i64, u.id.to_string()]).map_err(|e| Error::Protocol(e.to_string()))?; }
        Ok(())
    }

    pub async fn delete(&self, id: Uuid) -> Result<()> {
        let conn = self.conn.lock().await;
        conn.execute("DELETE FROM tunnels WHERE id=?1", params![id.to_string()])
            .map_err(|e| Error::Protocol(e.to_string()))?;
        Ok(())
    }
}
```

```rust
//! Manages one local-forward SSH tunnel: binds a TcpListener, accepts
//! connections, and splices each through a `channel_open_direct_tcpip`
//! channel on the shared russh Handle.

use crate::protocol::RusshHandle;
use russh::ChannelMsg;
use serde::Serialize;
use std::net::SocketAddr;
use tauri::{AppHandle, Emitter};
use tokio::io::{AsyncReadExt, AsyncWriteExt};
use tokio::net::TcpListener;
use tokio::sync::oneshot;
use uuid::Uuid;

pub const EV_TUNNEL_STATUS: &str = "tunnel:status";

#[derive(Clone, Serialize)]
pub struct TunnelStatusEvent {
    pub session_id: Uuid,
    pub rule_id: String,
    pub status: String,         // "active" | "error" | "closed"
    pub error: Option<String>,
}

/// Owns a running tunnel task. Dropping or calling `abort()` tears it down.
pub struct TunnelHandle {
    abort_tx: oneshot::Sender<()>,
}

impl TunnelHandle {
    pub fn abort(self) {
        let _ = self.abort_tx.send(());
    }
}

/// Spawn a local-forward tunnel task.
///
/// Binds `127.0.0.1:<local_port>`, then loops accepting TCP connections.
/// For each connection opens a `channel_open_direct_tcpip` SSH channel
/// to `remote_host:remote_port` and splices bytes bidirectionally.
/// Emits `tunnel:status` events on bind success/failure and on abort.
pub async fn spawn_tunnel(
    ssh: RusshHandle,
    session_id: Uuid,
    rule_id: String,
    local_port: u16,
    remote_host: String,
    remote_port: u16,
    app: AppHandle,
) -> Result<TunnelHandle, String> {
    let addr: SocketAddr = format!("127.0.0.1:{local_port}").parse().unwrap();
    let listener = TcpListener::bind(addr)
        .await
        .map_err(|e| format!("bind 127.0.0.1:{local_port}: {e}"))?;

    let (abort_tx, mut abort_rx) = oneshot::channel::<()>();

    let app2 = app.clone();
    let rule_id2 = rule_id.clone();

    // Emit "active" immediately after successful bind.
    let _ = app.emit(EV_TUNNEL_STATUS, TunnelStatusEvent {
        session_id, rule_id: rule_id.clone(), status: "active".into(), error: None,
    });

    tokio::spawn(async move {
        loop {
            tokio::select! {
                _ = &mut abort_rx => {
                    let _ = app2.emit(EV_TUNNEL_STATUS, TunnelStatusEvent {
                        session_id, rule_id: rule_id2.clone(), status: "closed".into(), error: None,
                    });
                    break;
                }
                accepted = listener.accept() => {
                    match accepted {
                        Ok((tcp, peer)) => {
                            let ssh_clone = Arc::clone(&ssh);
                            let rh = remote_host.clone();
                            let rp = remote_port;
                            let local_p = local_port;
                            let peer_port = peer.port() as u32;
                            tokio::spawn(async move {
                                let channel_result = ssh_clone
                                    .channel_open_direct_tcpip(
                                        rh, rp as u32,
                                        "127.0.0.1".to_string(), peer_port,
                                    )
                                    .await;
                                match channel_result {
                                    Ok(channel) => splice(tcp, channel).await,
                                    Err(e) => {
                                        log::warn!("direct-tcpip failed: {e}");
                                    }
                                }
                            });
                        }
                        Err(e) => {
                            log::warn!("tunnel accept error: {e}");
                            break;
                        }
                    }
                }
            }
        }
    });

    Ok(TunnelHandle { abort_tx })
}

async fn splice(tcp: tokio::net::TcpStream, mut channel: russh::Channel<russh::client::Msg>) {
    use std::sync::Arc;
    let (mut tcp_r, mut tcp_w) = tcp.into_split();

    let to_ssh = async {
        let mut buf = [0u8; 8192];
        loop {
            let n = tcp_r.read(&mut buf).await?;
            if n == 0 { break; }
            channel.data(&buf[..n]).await?;
        }
        let _ = channel.eof().await;
        Ok::<_, Box<dyn std::error::Error + Send + Sync>>(())
    };

    let from_ssh = async {
        while let Some(msg) = channel.wait().await {
            match msg {
                ChannelMsg::Data { data } => {
                    tcp_w.write_all(&data).await?;
                }
                ChannelMsg::Eof | ChannelMsg::Close => break,
                _ => {}
            }
        }
        Ok::<_, Box<dyn std::error::Error + Send + Sync>>(())
    };

    tokio::select! {
        _ = to_ssh => {}
        _ = from_ssh => {}
    }
}
```

```rust
impl SessionManager {
    pub async fn open_tunnel(
        &self,
        session_id: Uuid,
        rule_id: String,
        local_port: u16,
        remote_host: String,
        remote_port: u16,
        app: AppHandle,
    ) -> crate::error::Result<()> {
        let map = self.inner.lock().await;
        let lc = map.get(&session_id)
            .ok_or_else(|| crate::error::Error::Protocol("session not found".into()))?;
        let mut lc = lc.lock().await;
        let ssh = lc.ssh_handle.clone()
            .ok_or_else(|| crate::error::Error::Protocol("no SSH handle for this session".into()))?;
        let handle = crate::session::tunnel::spawn_tunnel(
            ssh, session_id, rule_id.clone(), local_port, remote_host, remote_port, app,
        ).await.map_err(|e| crate::error::Error::Protocol(e))?;
        lc.tunnels.insert(rule_id, handle);
        Ok(())
    }

    pub async fn close_tunnel(&self, session_id: Uuid, rule_id: &str) -> crate::error::Result<()> {
        let map = self.inner.lock().await;
        if let Some(lc) = map.get(&session_id) {
            let mut lc = lc.lock().await;
            if let Some(handle) = lc.tunnels.remove(rule_id) {
                handle.abort();
            }
        }
        Ok(())
    }

    pub async fn close_all_tunnels(&self, session_id: Uuid) {
        let map = self.inner.lock().await;
        if let Some(lc) = map.get(&session_id) {
            let mut lc = lc.lock().await;
            for (_, handle) in lc.tunnels.drain() {
                handle.abort();
            }
        }
    }
}
```
